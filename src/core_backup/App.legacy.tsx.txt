/** react-native-web의 Alert.alert는 아무 동작도 하지 않으므로 브라우저 대화상자로 대체한다. */
function installWebAlert(): void {
  if (!IS_WEB || typeof window === 'undefined') return;

  Alert.alert = (title, message, buttons) => {
    const text = [title, message].filter((part) => typeof part === 'string' && part.length > 0).join('\n\n');
    const list = Array.isArray(buttons) ? buttons : [];

    if (list.length <= 1) {
      window.alert(text);
      list[0]?.onPress?.();
      return;
    }

    // 확인/취소 2지선다: 취소 버튼이 있으면 그것을 거부 경로로, 없으면 마지막 버튼을 거부 경로로 본다.
    const cancelButton = list.find((button) => button.style === 'cancel') ?? list[list.length - 1];
    const confirmButton = list.find((button) => button !== cancelButton);
    const guide =
      confirmButton?.text && cancelButton?.text
        ? `\n\n▶ 확인: ${confirmButton.text} / 취소: ${cancelButton.text}`
        : '';
    if (window.confirm(`${text}${guide}`)) {
      confirmButton?.onPress?.();
    } else {
      cancelButton?.onPress?.();
    }
  };
}

installWebAlert();

}

/** 영문 시스템 오류 문구는 숨기고, 한글 안내 문구만 사용자에게 보여 준다. */
function describeError(error: unknown): string {
  const message = error instanceof Error ? error.message : typeof error === 'string' ? error : '';
  return /[가-힣]/.test(message) ? message : '잠시 후 다시 시도해 주세요.';
}

const memoryFlags = new Map<string, string>();

/** localStorage 기반 키-값 래퍼. 사용할 수 없는 환경(사생활 보호 모드 등)에서는 메모리로 폴백한다. */
const Storage = {
  getItemSync(key: string): string | null {
    try {
      return window.localStorage.getItem(key);
    } catch {
      return memoryFlags.get(key) ?? null;
    }
  },
  setItemSync(key: string, value: string): void {
    try {
      window.localStorage.setItem(key, value);
    } catch {
      memoryFlags.set(key, value);
    }
  },
  async getItem(key: string): Promise<string | null> {
    return Storage.getItemSync(key);
  },
  async setItem(key: string, value: string): Promise<void> {
    Storage.setItemSync(key, value);
  },
};

function readOnboardingFlag(): OnboardingFlag {
  try {
    const value = Storage.getItemSync(ONBOARDING_FLAG_KEY);
    return value === 'timeline' || value === 'done' ? value : null;
  } catch (e) {
    console.warn('온보딩 플래그 읽기 실패:', e);
    return null;
  }
}

function writeOnboardingFlag(value: Exclude<OnboardingFlag, null>): void {
  try {
    Storage.setItemSync(ONBOARDING_FLAG_KEY, value);
  } catch (e) {
    console.warn('온보딩 플래그 저장 실패:', e);
  }
}

/** 앱이 자동 생성한 샘플 프로필(사용자가 아직 정보를 입력하지 않은 상태)인지 판별한다. */
function isSampleProfile(profile: UserProfile): boolean {
  return (
    profile.birthDate === DEFAULT_BIRTH_DATE && profile.birthTime === DEFAULT_BIRTH_TIME && !profile.gender
  );
}

interface PartnerInfoPayload {
  alias: string;
  gender: string;
  birthDate: string;
  relation?: string;
  dayMaster?: string;
  dayBranch?: string;
}

function computeSajuFromProfile(
  birthDate: string,
  birthTime: string,
  calendarType: CalendarType
): SajuResult {
  const date = parseBirthDate(birthDate);
  if (!date) throw new Error('생년월일 형식이 올바르지 않습니다.');

  // 시간을 모르면 정오 기준으로 계산한다.
  const time = birthTime === BIRTH_TIME_UNKNOWN ? { hour: 12, minute: 0 } : parseBirthTime(birthTime);
  if (!time) throw new Error('태어난 시간 형식이 올바르지 않습니다.');

  return calculateSaju(date.year, date.month, date.day, time.hour, time.minute, calendarType !== 'lunar');
}

function buildPartnerInfo(partner: PartnerProfile): PartnerInfoPayload {
  const info: PartnerInfoPayload = {
    alias: partner.alias,
    gender: partner.gender ?? '',
    birthDate: partner.birthDate,
    relation: partner.relation || undefined,
  };
  try {
    const saju = computeSajuFromProfile(partner.birthDate, partner.birthTime, partner.calendarType);
    return { ...info, dayMaster: saju.dayMaster, dayBranch: saju.pillars.day.branch };
  } catch {
    // 상대 사주 계산에 실패해도 기본 정보만으로 분석은 계속 진행한다.
    return info;
  }
}

function parseStoredSaju(json: string | null | undefined): SajuResult | null {
  if (!json) return null;
  try {
    const value: unknown = JSON.parse(json);
    const saju = value as Partial<SajuResult> | null;
    if (
      saju &&
      typeof saju.dayMaster === 'string' &&
      saju.pillars?.day?.stem &&
      saju.pillars?.time?.stem &&
      saju.elementsRatio
    ) {
      return saju as SajuResult;
    }
    return null;
  } catch {
    return null;
  }
}

const MBTI_LETTER_LABEL: Record<string, string> = {
  E: '외향', I: '내향', S: '감각', N: '직관', T: '사고', F: '감정', J: '판단', P: '인식',
};

function buildMaskComment(sajuMbti: string, actualMbti: string, leakage: number): string {
  if (leakage <= 0) {
    return '선천 코어와 실제 성향이 일치합니다. 가면 없이 에너지를 100% 보존 중입니다.';
  }
  const innate: string[] = [];
  const persona: string[] = [];
  for (let i = 0; i < 4; i += 1) {
    const a = sajuMbti[i];
    const b = actualMbti[i];
    if (a && b && a !== b) {
      innate.push(`${MBTI_LETTER_LABEL[a] ?? a}(${a})`);
      persona.push(`${MBTI_LETTER_LABEL[b] ?? b}(${b})`);
    }
  }
  return `선천적 ${innate.join('·')}의 에너지를 지키기 위해 ${persona.join('·')} 페르소나를 가동 중. 일일 ${leakage}% 배터리 누수 발생`;
}

function getLeakageColor(leakage: number): string {
  if (leakage >= 75) return '#FF5555';
  if (leakage >= 50) return '#F1FA8C';
  return '#00F0FF';
}

      // 정오 기준으로 계산해 자시(23시~) 일진 경계 이슈를 피한다.
      const dayPillar = calculateSaju(yyyy, now.getMonth() + 1, now.getDate(), 12, 0, true).pillars.day;
      const dailyGanji = `${dayPillar.stem}${dayPillar.branch}`;

      await saveDailyLog(
        today,
        data.emotionElement,
        data.eventCategory,
        data.energyLevel,
        dailyGanji,
        data.shortMemo,
        undefined,
        data.tarotCard
      );
      Alert.alert(
        '⚡ 카드 기록 완료',
        `오늘의 오행 기운이 이 브라우저에 안전하게 저장되었습니다.${
          data.shortMemo ? '\n📝 오늘의 넋두리 한 줄도 함께 기록되었습니다.' : ''
        }`
      );
    } catch (e: any) {
      Alert.alert('저장 오류', `카드 기록을 저장하지 못했습니다. ${describeError(e)}`);
      throw e;
    }
  };

  const handleRequestAiAnalysis = async () => {
    if (!profile) {
      Alert.alert('안내', '프로필 정보가 필요합니다.');
      return;
    }

    setAiLoading(true);
    try {
      const now = new Date();
      const today = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(now.getDate()).padStart(2, '0')}`;

      const todayLog = await getDailyLog(today);
      if (!todayLog) {
        Alert.alert('안내', '먼저 오늘의 카드를 기록해 주세요.');
        return;
      }

      const sajuObj = JSON.parse(profile.sajuData);
      const currentAgent = AGENT_STORE[selectedAgent];

      const cardData = {
        emotionElement: todayLog.emotion_element,
        eventCategory: todayLog.event_category,
        energyLevel: todayLog.energy_level,
        shortMemo: todayLog.short_memo ?? '',
        dailyGanji: todayLog.daily_ganji,
        tarotCard: todayLog.tarotCard ?? '',
      };

      // 현재 배포된 게이트웨이는 diaryContent/emotionScore 필드로 프롬프트를 구성하므로 카드 요약을 함께 보낸다.
      const diaryContent =
        `[오늘의 3초 카드] 일진 ${cardData.dailyGanji} / 감정 오행 ${cardData.emotionElement} / ` +
        `사건 영역 ${cardData.eventCategory} / 에너지 잔량 ${cardData.energyLevel}%` +
        (cardData.tarotCard ? ` / 오늘의 타로 카드: ${cardData.tarotCard}` : '') +
        (cardData.shortMemo ? ` / 넋두리: "${cardData.shortMemo}"` : '') +
        (sajuMbti?.mbti
          ? ` / 선천 MBTI ${sajuMbti.mbti}` +
            (profile.actual_mbti ? `, 실제 MBTI ${profile.actual_mbti} (일치율 ${profile.mbti_sync ?? 100}%)` : '')
          : '') +
        ` [요청] 위 일진·오행 에너지·에너지 잔량·넋두리${cardData.tarotCard ? '·타로 카드' : ''}를 ` +
        `함께 엮어 해석하고, 당신의 페르소나 말투로 600~800자의 3단 점사 공식(간파, 당위성 부여, 단호한 단일 처방)으로 답하십시오.`;

      const partnerInfo =
        partnerProfile && cardData.eventCategory === LOVE_CATEGORY
          ? buildPartnerInfo(partnerProfile)
          : undefined;

      // 로컬 룰 엔진 연산(외부 통신 없음). 산출 실패 시 undefined라 JSON에서 제외되고 AI 요청은 그대로 진행된다.
      const structuralContext =
        getStructuralContext(profile.birthDate, now, {
          birthTime: profile.birthTime,
          calendarType: profile.calendarType ?? 'solar',
          gender: profile.gender ?? null,
        }) ?? undefined;

      const response = await fetch('https://saju-edge-gateway.cybersaju.workers.dev', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          agentType: selectedAgent,
          systemInstruction: currentAgent.systemInstruction,
          dayMaster: sajuObj?.dayMaster,
          pillars: sajuObj?.pillars,
          elementsRatio: sajuObj?.elementsRatio,
          coreMbti: sajuMbti?.mbti,
          actualMbti: profile.actual_mbti,
          mbtiSync: profile.mbti_sync,
          gender: profile.gender ?? undefined,
          userLore: profile.userLore ?? '',
          partnerInfo,
          // 사용자가 대운 동기화를 확정한 경우에만 보낸다. 미확정이면 undefined라 JSON에서 제외된다.
          syncRatio: lifeOnboarding.isSynced ? lifeOnboarding.syncRatio : undefined,
          interests: lifeOnboarding.isSynced ? lifeOnboarding.interests : undefined,
          structuralContext,
          cardData,
          tarotCard: cardData.tarotCard,
          diaryContent,
          emotionScore: Math.max(1, Math.min(5, Math.round(cardData.energyLevel / 20))),
        }),
      });

      const data = await response.json();
      if (!response.ok) {
        throw new Error(data.error || JSON.stringify(data));
      }

      const resultText =
        data.analysis ||
        data.feedback ||
        (typeof data === 'string' ? data : JSON.stringify(data));

      setAnalysisResult(resultText);

      await saveDailyLog(
        today,
        todayLog.emotion_element,
        todayLog.event_category,
        todayLog.energy_level,
        todayLog.daily_ganji,
        todayLog.short_memo ?? undefined,
        `[${currentAgent.name}] ${resultText}`,
        todayLog.tarotCard
      );

      Alert.alert('인공지능 점사 완료', `[${currentAgent.name}]의 처방이 오늘의 카드 기록에 저장되었습니다.`);
    } catch (err: any) {
      Alert.alert(
        '통신 오류',
        `인공지능 도사와 연결하지 못했습니다. 네트워크 상태를 확인하신 뒤 다시 시도해 주세요. ${describeError(err)}`
      );
    } finally {
      setAiLoading(false);
    }
  };

  // 프로필 로드 여부와 관계없이 릴스 화면을 먼저 그린다. (로드 전에는 오늘 일진 기반 기본 문구)
  const sajuObj = profile ? parseStoredSaju(profile.sajuData) : null;
  const todayKey = new Date().toDateString();
  const reelsContent = useMemo(
    () =>
      buildReelsContent(sajuObj, new Date(), {
        aiText: analysisResult,
        profileId: profile?.id ?? null,
        profileName: profile?.name ?? null,
      }),
    // sajuObj는 profile.sajuData에서 파생되므로 원본 문자열을 의존성으로 쓴다.
    // eslint-disable-next-line react-hooks/exhaustive-deps
    [profile?.sajuData, profile?.id, profile?.name, analysisResult, todayKey]
  );
  const currentAgent = AGENT_STORE[selectedAgent];

  const savedActualMbti = profile?.actual_mbti ?? null;
  const actualMbti = isValidMbti(savedActualMbti) ? savedActualMbti : null;
  const mbtiSyncResult =
    sajuMbti?.mbti && actualMbti ? calculateMbtiSync(sajuMbti.mbti, actualMbti) : null;
  const leakageColor = getLeakageColor(mbtiSyncResult?.energyLeakage ?? 0);
  const batteryRemain = 100 - (mbtiSyncResult?.energyLeakage ?? 0);

  return (
    <View style={[styles.root, WEB_VIEWPORT_STYLE]}>
      {/* 영구 기본 메인 뷰: 프로필 로드 여부와 관계없이 항상 풀스크린 릴스를 그린다. */}
      <ReelsFortuneView
        element={reelsContent.element}
        elementName={reelsContent.elementName}
        dayPillarText={reelsContent.dayPillarText}
        keyword={reelsContent.keyword}
        fortuneText={reelsContent.fortuneText}
        luckyItem={reelsContent.luckyItem}
        luckyReason={reelsContent.luckyReason}
        profileLabel={reelsContent.profileLabel}
        onSettingsPress={() => setDashboardVisible(true)}
        onTarotPress={handleReelsTarot}
        onMatchPress={handleReelsMatch}
        onSharePress={handleReelsShare}
        onOpenDashboard={() => setDashboardVisible(true)}
      />

      {/* 기존 진단 대시보드: 메인 화면을 가리지 않고 하단 시트로만 열린다. */}
      <Modal
        visible={dashboardVisible}
        transparent
        animationType="slide"
        statusBarTranslucent
        onRequestClose={() => setDashboardVisible(false)}
      >
        <View style={styles.sheetBackdrop}>
          <Pressable
            style={styles.sheetDismiss}
            onPress={() => setDashboardVisible(false)}
            accessibilityRole="button"
            accessibilityLabel="대시보드 닫기"
          />
          <View style={[styles.sheet, { height: Math.round(windowHeight * 0.88) }]}>
            <View style={styles.sheetHandle} />
            <View style={styles.sheetHeader}>
              <Text style={styles.sheetTitle}>⚙️ 사주 대시보드</Text>
              <TouchableOpacity
                onPress={() => setDashboardVisible(false)}
                activeOpacity={0.7}
                accessibilityRole="button"
                accessibilityLabel="대시보드 닫기"
                hitSlop={{ top: 8, bottom: 8, left: 8, right: 8 }}
                style={styles.sheetCloseBtn}
              >
                <Text style={styles.sheetCloseText}>✕</Text>
              </TouchableOpacity>
            </View>
    <ScrollView
      ref={scrollRef}
      style={styles.scrollView}
      contentContainerStyle={[styles.container, styles.sheetContent]}
    >
      {/* 2단 반응형 헤더 & 명확한 [💾 DB 백업] 버튼 */}
      <View style={styles.headerRow}>
        <View style={styles.headerTitleBox}>
          <Text style={styles.headerTitle}>옥동자의 사이버 작두</Text>
          <Text style={styles.headerSubtitle}>내 기기에만 저장되는 사주 점사</Text>
        </View>
        <View style={styles.headerActions}>
          <TouchableOpacity
            style={styles.backupHeaderBtn}
            onPress={() => setBackupVisible(true)}
            activeOpacity={0.7}
          >
            <Text style={styles.backupHeaderBtnText}>💾 데이터보관</Text>
          </TouchableOpacity>
          <TouchableOpacity
            style={styles.settingsHeaderBtn}
            onPress={() => {
              setOnboarding(false);
              setProfileModalVisible(true);
            }}
            activeOpacity={0.7}
            accessibilityRole="button"
            accessibilityLabel="운명 정보 설정"
          >
            <Text style={styles.settingsHeaderBtnText}>⚙️</Text>
          </TouchableOpacity>
        </View>
      </View>

      <TouchableOpacity
        style={styles.timelineBtn}
        onPress={() => setTimelineVisible(true)}
        activeOpacity={0.8}
        accessibilityRole="button"
        accessibilityLabel="일생 대운 시간여행 열기"
      >
        <Text style={styles.timelineBtnText}>
          {lifeOnboarding.isSynced ? '🪐 일생 대운 · 시간여행' : '🪐 일생 대운 확인'}
        </Text>
        {lifeOnboarding.isSynced && (
          <View style={styles.timelineBadge}>
            <Text style={styles.timelineBtnSub}>🪐 대운 일치율 {lifeOnboarding.syncRatio}%</Text>
          </View>
        )}
      </TouchableOpacity>

      <View style={styles.card}>
        <Text style={styles.cardTitle}>[내 사주 정보 · 이 기기에 저장]</Text>
        <Text style={styles.text}>식별자: #{profile?.id} | 사용자: {profile?.name}</Text>
        <Text style={styles.text}>
          생년월일: {profile?.birthDate} {profile?.birthTime}
          {profile?.calendarType === 'lunar' ? ' (음력)' : ' (양력)'}
          {profile?.gender ? ` · ${profile.gender === 'male' ? '남성' : '여성'}` : ''}
        </Text>
        {sajuObj && (
          <Text style={styles.sajuPillars}>
            {sajuObj.pillars.year.stem}{sajuObj.pillars.year.branch}년{' '}
            {sajuObj.pillars.month.stem}{sajuObj.pillars.month.branch}월{' '}
            {sajuObj.pillars.day.stem}{sajuObj.pillars.day.branch}일{' '}
            {sajuObj.pillars.time.stem}{sajuObj.pillars.time.branch}시 (일간 {sajuObj.dayMaster})
          </Text>
        )}
      </View>

      {!!sajuMbti?.mbti && (
        <View style={styles.mbtiCard}>
          <Text style={styles.mbtiCardTitle}>[타고난 성향 · 사주로 본 선천 MBTI]</Text>
          <Text style={styles.mbtiLetters}>{sajuMbti.mbti}</Text>
          <Text style={styles.mbtiDescription}>{sajuMbti.description}</Text>

          <View style={styles.mbtiCalibrateRow}>
            <Text style={styles.mbtiCalibrateHint}>
              {actualMbti ? `실제 MBTI: ${actualMbti}` : '실제 MBTI와 다른가요?'}
            </Text>
            <TouchableOpacity
              style={styles.mbtiCalibrateBtn}
              onPress={() => setMbtiPickerVisible(true)}
              activeOpacity={0.7}
            >
              <Text style={styles.mbtiCalibrateBtnText}>MBTI 보정</Text>
            </TouchableOpacity>
          </View>

          {mbtiSyncResult && actualMbti && (
            <View style={styles.leakageBox}>
              <Text style={[styles.leakageTitle, { color: leakageColor }]}>
                [사회적 가면 누수율: {mbtiSyncResult.energyLeakage}%]
              </Text>
              <View style={styles.batteryRow}>
                <View style={[styles.batteryBody, { borderColor: leakageColor, shadowColor: leakageColor }]}>
                  {[0, 1, 2, 3].map((i) => {
                    const lit = (i + 1) * 25 <= batteryRemain;
                    return (
                      <View
                        key={i}
                        style={[
                          styles.batteryCell,
                          lit && { backgroundColor: leakageColor, shadowColor: leakageColor, shadowOpacity: 0.9, shadowRadius: 6 },
                        ]}
                      />
                    );
                  })}
                </View>
                <View style={[styles.batteryCap, { backgroundColor: leakageColor }]} />
              </View>
              <Text style={styles.leakageSync}>
                선천 {sajuMbti.mbti} ↔ 실제 {actualMbti} · 일치율 {mbtiSyncResult.syncRate}%
              </Text>
              <Text style={styles.leakageComment}>
                {buildMaskComment(sajuMbti.mbti, actualMbti, mbtiSyncResult.energyLeakage)}
              </Text>
            </View>
          )}
        </View>
      )}

      {celebrityResult?.matchedCelebrity && (
        <View style={styles.celebCard}>
          <Text style={styles.celebCardTitle}>[세계 유명인 사주 거울 · 영혼의 주파수]</Text>
          <View style={styles.celebSyncBadge}>
            <Text style={styles.celebSyncText}>⚡ 사주 일치율 {celebrityResult.syncRate}%</Text>
          </View>

          <Text style={styles.celebName}>{celebrityResult.matchedCelebrity.name}</Text>
          {!!celebrityResult.matchedCelebrity.title && (
            <Text style={styles.celebTitle}>{celebrityResult.matchedCelebrity.title}</Text>
          )}
          {!!celebrityResult.matchedCelebrity.archetype && (
            <Text style={styles.celebArchetype}>#{celebrityResult.matchedCelebrity.archetype}</Text>
          )}

          <View style={styles.celebReasonBox}>
            <Text style={styles.celebReasonLabel}>명리학 일치 사유</Text>
            <Text style={styles.celebReasonText}>{celebrityResult.matchReason}</Text>
          </View>

          {!!celebrityResult.matchedCelebrity.quote && (
            <Text style={styles.celebQuote}>“{celebrityResult.matchedCelebrity.quote}”</Text>
          )}

          <TouchableOpacity
            style={styles.celebShareBtn}
            onPress={() => setShareModalVisible(true)}
            activeOpacity={0.7}
          >
            <Text style={styles.celebShareBtnText}>[📸 인스타 스토리 공유 카드 생성]</Text>
          </TouchableOpacity>
        </View>
      )}

      {sajuObj?.elementsRatio && (
        <ElementCircuit elementsRatio={sajuObj.elementsRatio} />
      )}

      <View style={styles.agentCard}>
        <Text style={styles.agentCardTitle}>[인공지능 도사 고르기]</Text>
        <TouchableOpacity
          style={styles.personaTrigger}
          onPress={() => setPersonaModalVisible(true)}
          activeOpacity={0.7}
        >
          <Text style={styles.personaTriggerEmoji}>{currentAgent.emoji}</Text>
          <View style={styles.personaTriggerTextBox}>
            <Text style={styles.personaTriggerName}>{currentAgent.name}</Text>
            <Text style={styles.personaTriggerTitle} numberOfLines={1}>
              {currentAgent.title}
            </Text>
          </View>
          <Text style={styles.personaTriggerChange}>▼ 변경</Text>
        </TouchableOpacity>
        <View style={styles.agentMetaBox}>
          <Text style={styles.agentMetaTone}>{currentAgent.tone}</Text>
        </View>
      </View>

      <View
        style={styles.dailyDeckWrap}
        onLayout={(e) => {
          deckYRef.current = e.nativeEvent.layout.y;
        }}
      >
        <DailyCardDeck onSave={handleSaveDailyCard} onEventCategorySelect={handleEventCategorySelect} />
        {partnerProfile && (
          <View style={styles.partnerChip}>
            <Text style={styles.partnerChipText} numberOfLines={1}>
              [대상: {partnerProfile.relation || partnerProfile.alias} ({partnerProfile.birthDate})]
            </Text>
            <TouchableOpacity
              onPress={() => setPartnerModalVisible(true)}
              activeOpacity={0.7}
              accessibilityRole="button"
              accessibilityLabel="상대방 정보 변경"
              hitSlop={{ top: 8, bottom: 8, left: 8, right: 8 }}
            >
              <Text style={styles.partnerChipChange}>[변경]</Text>
            </TouchableOpacity>
          </View>
        )}
      </View>

      <TouchableOpacity
        style={[styles.aiButton, aiLoading && styles.buttonDisabled]}
        onPress={handleRequestAiAnalysis}
        disabled={aiLoading}
      >
        {aiLoading ? (
          <ActivityIndicator color="#090D16" />
        ) : (
          <Text style={styles.aiButtonText}>
            ⚡ [{currentAgent.name}] 오행 교차 분석 요청
          </Text>
        )}
      </TouchableOpacity>

      {analysisResult && (
        <View style={styles.analysisCard}>
          <Text style={styles.analysisTitle}>[인공지능 점사 결과 · {currentAgent.name}]</Text>
          <Text style={styles.analysisContent}>{analysisResult}</Text>
        </View>
      )}
    </ScrollView>
            <View style={styles.quickMenu}>
              {QUICK_MENU_ITEMS.map((item) => (
                <TouchableOpacity
                  key={item.key}
                  style={styles.quickItem}
                  onPress={() => handleQuickMenu(item.key)}
                  activeOpacity={0.75}
                  accessibilityRole="button"
                  accessibilityLabel={item.label}
                >
                  <Text style={styles.quickIcon}>{item.icon}</Text>
                  <Text style={styles.quickLabel} numberOfLines={1}>
                    {item.label}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
          </View>
        </View>
      </Modal>

      <BackupModal
        visible={backupVisible}
        onClose={() => setBackupVisible(false)}
        onExport={exportBackupJson}
        onImport={importBackupJson}
        onRefresh={loadData}
      />

      <MbtiPickerModal
        visible={mbtiPickerVisible}
        selected={actualMbti}
        innateMbti={sajuMbti?.mbti}
        onClose={() => setMbtiPickerVisible(false)}
        onSelect={handleSelectActualMbti}
      />

      <PersonaPickerModal
        visible={personaModalVisible}
        selected={selectedAgent}
        onClose={() => setPersonaModalVisible(false)}
        onSelect={(agent) => {
          setSelectedAgent(agent);
          setPersonaModalVisible(false);
        }}
      />

      <ProfileSettingModal
        visible={profileModalVisible}
        onClose={() => {
          setProfileModalVisible(false);
          setOnboarding(false);
        }}
        isOnboarding={onboarding}
        initialValues={
          profile && !onboarding
            ? {
                birthDate: profile.birthDate,
                birthTime: profile.birthTime,
                gender: profile.gender ?? undefined,
                calendarType: profile.calendarType ?? 'solar',
                userLore: profile.userLore ?? '',
              }
            : null
        }
        onSave={handleSaveProfile}
        onSaved={handleProfileSaved}
      />

      <PartnerInputModal
        visible={partnerModalVisible}
        onClose={() => setPartnerModalVisible(false)}
        initialValues={partnerProfile}
        onSaved={handlePartnerSaved}
      />

      <LifeTimelineModal
        visible={timelineVisible}
        onClose={handleTimelineClose}
        isOnboarding={onboardingStage === 'timeline'}
        birthDate={profile?.birthDate ?? ''}
        birthTime={profile?.birthTime ?? ''}
        gender={profile?.gender ?? null}
        calendarType={profile?.calendarType ?? 'solar'}
        initialSyncRatio={lifeOnboarding.syncRatio}
        initialInterests={lifeOnboarding.interests}
        isSynced={lifeOnboarding.isSynced}
        onSynced={handleTimelineSynced}
        onRequestProfile={handleTimelineRequestProfile}
      />

      <CelebrityShareModal
        visible={shareModalVisible}
        onClose={() => setShareModalVisible(false)}
        celebrityResult={celebrityResult}
        userDayMaster={sajuObj?.dayMaster ?? ''}
      />
    </View>
  );
}
